### 2.2 SPU（产品本体）

> SPU = 货物单元 + 容量/规格 + 国标型号，**不含品牌**

```sql
CREATE TABLE spu (
    spu_id          VARCHAR(32)    PRIMARY KEY,
    gu_id           VARCHAR(32)    NOT NULL,        -- 关联货物单元
    spu_name        VARCHAR(200)   NOT NULL,         -- 手提式干粉灭火器 4kg
    model_code      VARCHAR(100)   NOT NULL,         -- 国标型号 MFZ/ABC4
    capacity        DECIMAL(10,2),                   -- 充装量数值 4
    capacity_unit   VARCHAR(10),                     -- kg / L
    fire_class      VARCHAR(50),                     -- 灭火级别 2A40B
    pressure_type   TINYINT,                         -- 1储压式 2贮气式
    valid_years     INT            DEFAULT 5,         -- 有效期年限
    description     TEXT,
    status          TINYINT        DEFAULT 1,         -- 1启用 0停用
    created_at      DATETIME
);
```

### 2.3 SKU（最小库存交易单元）

> SKU = SPU + 品牌，是所有出入库、采购、销售单据的操作对象

```sql
CREATE TABLE sku (
    sku_id          VARCHAR(32)    PRIMARY KEY,
    spu_id          VARCHAR(32)    NOT NULL,
    brand           VARCHAR(100)   NOT NULL,         -- 金盾、淮海、洪湖
    sku_name        VARCHAR(200)   NOT NULL,         -- 金盾 手提式干粉灭火器 4kg
    cert_3c_no      VARCHAR(100),                    -- 3C证书编号（必填）
    barcode         VARCHAR(64),                     -- 条形码
    weight_net      DECIMAL(10,3),                   -- 净重 kg
    weight_gross    DECIMAL(10,3),                   -- 毛重 kg
    sale_price      DECIMAL(10,2),                   -- 默认销售价
    cost_price      DECIMAL(10,2),                   -- 最近采购成本
    stock_qty       INT            DEFAULT 0,         -- 当前库存（汇总）
    stock_warn_min  INT            DEFAULT 10,        -- 库存下限预警
    status          TINYINT        DEFAULT 1,
    created_at      DATETIME
);
```

### 2.4 供应商与采购价目表

```sql
-- 供应商
CREATE TABLE supplier (

    address         VARCHAR(255),

-- 采购价目表（SKU + 供应商 → 价格）
CREATE TABLE purchase_price (
    id              BIGINT        PRIMARY KEY AUTO_INCREMENT,
    sku_id          VARCHAR(32)   NOT NULL,
    supplier_id     VARCHAR(32)   NOT NULL,
    price           DECIMAL(10,2) NOT NULL,   -- 含税采购价

    valid_from      DATE,
    valid_to        DATE,
    is_default      TINYINT       DEFAULT 0,  -- 是否默认供应商
    UNIQUE KEY uk_sku_supplier (sku_id, supplier_id)
);
```

## 三、仓库与批次设计

### 3.1 批次表（消防器材必须管批次）

> 灭火器有效期5年，批次管理是合规核心

```sql
CREATE TABLE stock_batch (
    batch_id        VARCHAR(64)   PRIMARY KEY,
    sku_id          VARCHAR(32)   NOT NULL,
    warehouse_id    VARCHAR(32)   NOT NULL,
    produce_date    DATE          NOT NULL,   -- 生产日期
    expire_date     DATE          NOT NULL,   -- 过期日期（produce_date + valid_years）
    batch_no        VARCHAR(64),              -- 生产批号
    purchase_order_id VARCHAR(32),            -- 来源采购单
    qty_in          INT           NOT NULL,   -- 入库数量
    qty_current     INT           NOT NULL,   -- 当前剩余数量
    cost_price      DECIMAL(10,2),            -- 该批次采购价
    status          TINYINT,      -- 1正常 2临期（距到期<6个月）3已过期
    created_at      DATETIME
);
```

### 3.2 库存出入库流水

```sql
CREATE TABLE stock_log (
    log_id          BIGINT        PRIMARY KEY AUTO_INCREMENT,
    sku_id          VARCHAR(32)   NOT NULL,
    batch_id        VARCHAR(64),
    warehouse_id    VARCHAR(32),
    biz_type        TINYINT,      -- 1采购入库 2销售出库 3盘点 4调拨
    biz_order_id    VARCHAR(32),  -- 关联单据ID
    qty_change      INT,          -- 正为入，负为出
    qty_before      INT,
    qty_after       INT,
    operate_time    DATETIME,
    operator        VARCHAR(50)
);
```

## 六、预警机制（消防行业刚需）

```sql
-- 系统需要定时跑以下预警逻辑

-- 1. 库存下限预警
SELECT s.sku_id, s.sku_name, s.stock_qty, s.stock_warn_min
FROM sku s
WHERE s.stock_qty <= s.stock_warn_min AND s.status = 1;

-- 2. 临期批次预警（6个月内到期）
SELECT b.batch_id, s.sku_name, b.expire_date, b.qty_current,
       DATEDIFF(b.expire_date, NOW()) AS days_left
FROM stock_batch b
JOIN sku s ON b.sku_id = s.sku_id
WHERE b.expire_date BETWEEN NOW() AND DATE_ADD(NOW(), INTERVAL 6 MONTH)
  AND b.qty_current > 0;

-- 3. 已过期库存预警
SELECT b.batch_id, s.sku_name, b.expire_date, b.qty_current
FROM stock_batch b
JOIN sku s ON b.sku_id = s.sku_id
WHERE b.expire_date < NOW() AND b.qty_current > 0;
```
